**POST** `/orgs/{organizationId}/members/{userId}/sso`

**Parameters**

<ParamField header="Authorization" param-type="string" required={true} />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="userId" param-type="string" required={true}>
  The unique ID of the User
</ParamField>

**Responses**

`200` The user has been added as an SSO member of the organization.

<ResponseField name="object" field-type="string" required={true} enum="member">
  Type of Object, always equals to "member"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the user.
</ResponseField>

<ResponseField name="role" field-type="string" required={true}>
  The role of a member in an organization, null for guests
</ResponseField>

<ResponseField name="user" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="object" field-type="string" required={true} enum="user">
      Type of Object, always equals to "user"
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Unique identifier for the user
    </ResponseField>

    <ResponseField name="displayName" field-type="string" required={true}>
      Full name for the user
    </ResponseField>

    <ResponseField name="email" field-type="string">
      Email address of the user
    </ResponseField>

    <ResponseField name="photoURL" field-type="string">
      URL of the user's profile picture
    </ResponseField>

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the object

      <Expandable title="Object Properties">
        <ResponseField name="location" field-type="string" required={true}>
          URL of the user in the API
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="disabled" field-type="string" required={true}>
  Why the member is disabled, or false when the member is active. A disabled member cannot access the organization or its content.
</ResponseField>

<ResponseField name="joinedAt" field-type="string" required={true} />

<ResponseField name="lastSeenAt" field-type="string" />

<ResponseField name="sso" field-type="boolean" required={true}>
  Whether the user can login with SSO.
</ResponseField>

<ResponseField name="spaces" field-type="number" required={true} />

<ResponseField name="teams" field-type="number" required={true} />

<ResponseField name="sites" field-type="number" required={true} />
