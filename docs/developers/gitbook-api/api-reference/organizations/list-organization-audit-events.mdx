Lists recorded changes for an activated organization. Access is restricted to organization administrators and GitBook staff, and results are ordered by occurrence time, newest first. An active Enterprise plan is required except for GitBook's own organization.

**GET** `/orgs/{organizationId}/audit-events`

**Parameters**

<ParamField header="Authorization" param-type="string" required={true} />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="integer">
  Number of events to return, up to 100.
</ParamField>

<ParamField query="from" param-type="string">
  Include events occurring at or after this time.
</ParamField>

<ParamField query="to" param-type="string">
  Include events occurring at or before this time.
</ParamField>

<ParamField query="action" param-type="string">
  Filter by an exact audit action name.
</ParamField>

<ParamField query="actor" param-type="string">
  Filter by an exact stable actor identifier.
</ParamField>

<ParamField query="target" param-type="string">
  Filter by an exact stable target identifier.
</ParamField>

<ParamField query="includeFacets" param-type="boolean">
  Include matching counts for action and actor options, excluding the option's own filter.
</ParamField>

<ParamField query="actions" param-type="string[]">
  Include any of these action names. Combined with other filters using AND.
</ParamField>

<ParamField query="actors" param-type="string[]">
  Include any of these stable actor identifiers, including system actors. Combined with other filters using AND.
</ParamField>
