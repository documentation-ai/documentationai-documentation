Returns the full allowlisted payload for one recorded organization change. Access is restricted to organization administrators and GitBook staff. The organization must be activated and on an active Enterprise plan, except for GitBook's own organization.

**GET** `/orgs/{organizationId}/audit-events/{auditEventId}`

**Parameters**

<ParamField header="Authorization" param-type="string" required={true} />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="auditEventId" param-type="string" required={true}>
  Unique identifier of the audit event.
</ParamField>

**Responses**

`200` The audit event.

<ResponseField name="object" field-type="string" required={true} enum="audit-event" />

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier of the audit event.
</ResponseField>

<ResponseField name="occurredAt" field-type="string" required={true} />

<ResponseField name="action" field-type="string" required={true}>
  Stable action name describing the recorded change.
</ResponseField>

<ResponseField name="actor" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="type" field-type="string" required={true} enum="user,integration,installation,system,support">
      Kind of actor that initiated the change.
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Stable identifier of the actor.
    </ResponseField>

    <ResponseField name="label" field-type="string">
      Display label captured when the event was recorded.
    </ResponseField>

    <ResponseField name="impersonator" field-type="object">
      GitBook staff member acting as this user. Present only for recorded impersonation; older events may not have this information. The actor ID remains the effective user for filtering.

      <Expandable title="Object Properties">
        <ResponseField name="id" field-type="string" required={true}>
          Stable ID of the impersonating staff member.
        </ResponseField>

        <ResponseField name="label" field-type="string">
          Staff display name captured when the event was recorded, if available.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="source" field-type="string" required={true} enum="ui,api,integration,webhook,workflow,script,support">
  Entry point that initiated the change.

<ResponseField name="target" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="type" field-type="string" required={true}>
      Kind of resource affected by the change.
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Stable identifier of the affected resource.
    </ResponseField>

    <ResponseField name="label" field-type="string">
      Display label captured when the event was recorded.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>

</ResponseField>

  Version of the action payload schema.
