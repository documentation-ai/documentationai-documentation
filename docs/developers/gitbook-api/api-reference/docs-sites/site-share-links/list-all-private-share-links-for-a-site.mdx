Returns a paginated list of share links configured for the site. Share links grant access to a private site without requiring full authentication. Use this to audit existing share links, check their active status, or find a link ID before updating or revoking it.

**GET** `/orgs/{organizationId}/sites/{siteId}/share-links`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="search" param-type="string">
  Search share links by name or key
</ParamField>

**Responses**

`200` OK

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="object" field-type="string" required={true} enum="share-link">
      Type of Object, always equals to "share-link"
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Unique identifier for the share-link
    </ResponseField>

    <ResponseField name="createdAt" field-type="string" required={true} />

    <ResponseField name="name" field-type="string">
      Name of the share link
    </ResponseField>

    <ResponseField name="active" field-type="boolean" />

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the object

      <Expandable title="Object Properties">
        <ResponseField name="published" field-type="string">
          URL of the published version of the share-link.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
