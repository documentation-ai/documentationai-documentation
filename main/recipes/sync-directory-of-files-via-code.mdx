```node-simple Node
const fs = require('fs');
const sdk = require('api')('https://docs.readme.com/developers/openapi/5e1a4f6284248b0018149174');

sdk.auth("API_KEY");

const createDocs = async () => {
  const categories = await sdk.getCategories({ perPage: "10", page: "1" });
  const docs = fs.readdirSync("./docs");
  
  for (const doc of docs) {
    const docContent = fs.readFileSync(doc);

    sdk
      .createDoc({
        title: doc,
        body: docContent,
        category: categories[0]._id,
      })
      .then((res) => {
        console.log(res);
      });
	}
};

createDocs();

```
